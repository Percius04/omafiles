# ---------------------------------------------------------------------------
# backend/ : Omafiles.Backend QML module as a SHARED PLUGIN (Phase 5.B)
# ---------------------------------------------------------------------------
# Before (5.A) the backend was compiled INSIDE the standalone executable, which
# prevented Quickshell (a preinstalled binary that only loads QML) from
# using it. Now it's built as a shared library with a QML plugin: a
# single .so + qmldir in build/qml/Omafiles/Backend/, loadable by import
# path both by the standalone and by Quickshell -- the same file
# for both, without duplication. See BACKEND_DESIGN.md, step 5.B.
#
qt_add_library(omafiles-backend SHARED)
qt_add_qml_module(omafiles-backend
  URI Omafiles.Backend
  VERSION 1.0
  PLUGIN_TARGET omafiles-backend
  OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/qml/Omafiles/Backend
  SOURCES
    backend/ProcessRunner.h
    backend/ProcessRunner.cpp
    backend/ProcessWatcher.h
    backend/ProcessWatcher.cpp
    backend/Env.h
    backend/Env.cpp
    backend/Detached.h

# The autogenerated QML type registration does #include <ProcessRunner.h>
# (angle brackets) -- backend/ has to be in the include path or
# __has_include fails and the type is left undeclared.
target_include_directories(omafiles-backend PRIVATE backend ${GIO_INCLUDE_DIRS})

target_link_libraries(omafiles-backend PRIVATE Qt6::Core Qt6::Gui Qt6::Qml Qt6::Quick Qt6::Pdf Qt6::DBus Qt6::Network ${GIO_LIBRARIES})

# ---------------------------------------------------------------------------
# app : Qt6 frontend
# ---------------------------------------------------------------------------
qt_add_executable(omafiles-standalone main.cpp)

# The standalone no longer compiles the backend inside: it loads it by import path
# just like Quickshell will (main.cpp uses OMAFILES_QML_IMPORT_DIR). It only
# needs the plugin to be built before starting.
add_dependencies(omafiles-standalone omafiles-backend)

target_link_libraries(omafiles-standalone PRIVATE
  Qt6::Core
  Qt6::Gui
  Qt6::Qml
  Qt6::Quick
  Qt6::QuickControls2
  Qt6::DBus
  Qt6::Network
)

# Paths that main.cpp needs to know at compile time (Phase 29, decoupling from the
# repo). resolveResourceDir() tries in order: OMAFILES_SOURCE_DIR (dev tree)
# -> OMAFILES_DATA_INSTALL_DIR/omafiles (installed) -> $XDG_DATA_HOME/omafiles.
#   OMAFILES_SOURCE_DIR      : development QML tree (loaded from source).
#   OMAFILES_QML_IMPORT_DIR  : Omafiles.Backend plugin in build/ (dev).
#   OMAFILES_QML_INSTALL_DIR : installed Omafiles.Backend plugin (stable).
#   OMAFILES_DATA_INSTALL_DIR: XDG_DATA root where the resources are installed.
set(OMAFILES_DATA_INSTALL_DIR "$ENV{HOME}/.local/share"
    CACHE PATH "XDG_DATA root where the Omafiles resource tree is installed")
# Stable path of the backend module (defined here, BEFORE the compile
# definitions that reference it; the install() that uses it lives further down).
set(OMAFILES_QML_INSTALL_DIR "$ENV{HOME}/.local/lib/qt6/qml"
    CACHE PATH "Stable path where the Omafiles.Backend QML module is installed")
target_compile_definitions(omafiles-standalone PRIVATE
  OMAFILES_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}"
  OMAFILES_QML_IMPORT_DIR="${CMAKE_BINARY_DIR}/qml"
  OMAFILES_QML_INSTALL_DIR="${OMAFILES_QML_INSTALL_DIR}"
  OMAFILES_DATA_INSTALL_DIR="${OMAFILES_DATA_INSTALL_DIR}"
)

# ---------------------------------------------------------------------------
# Stable deploy of the module (Phase 5.B.1)
# ---------------------------------------------------------------------------
# The real Omarchy shell (Quickshell under uwsm) loads Omafiles.Backend
# via QML_IMPORT_PATH, defined persistently in
# ~/.config/environment.d/omafiles-backend.conf. That variable points to a
# STABLE user path, not to build/ (ephemeral, deleted on clean).
# `cmake --install build` copies the module there. See BACKEND_DESIGN.md 4.
# (OMAFILES_QML_INSTALL_DIR is defined above, next to the compile defs.)
install(DIRECTORY "${CMAKE_BINARY_DIR}/qml/Omafiles"
        DESTINATION "${OMAFILES_QML_INSTALL_DIR}")

# ---------------------------------------------------------------------------
# Installation of the standalone binary (Phase 25: official Qt6 frontend)
# ---------------------------------------------------------------------------
# The executable is installed RENAMED to `omafiles`. User builds default to
# ~/.local/bin; package builds override this with /usr/bin. Runtime QML, scripts,
# assets, and integration descriptors are installed below, so the source checkout
# is not needed after installation. User defaults remain explicit and reversible
# through scripts/install-integrations.sh --enable/--disable.
set(OMAFILES_BIN_INSTALL_DIR "$ENV{HOME}/.local/bin"
    CACHE PATH "Path where the omafiles binary is installed")
install(PROGRAMS "$<TARGET_FILE:omafiles-standalone>"
        DESTINATION "${OMAFILES_BIN_INSTALL_DIR}"
        RENAME omafiles)

# ---------------------------------------------------------------------------
# Installation of the RESOURCE tree (Phase 29: independent from the repo)
# ---------------------------------------------------------------------------
# main.cpp (resolveResourceDir) loads the QML and locates the .sh scripts from
# here when the repo no longer exists. The layout is PRESERVED: state/Paths.qml
# expects <resourceDir>/<script>.sh at the root and <resourceDir>/scripts/... .
set(OMAFILES_RES_DEST "${OMAFILES_DATA_INSTALL_DIR}/omafiles")

# Complete QML tree that the engine loads at runtime.
install(DIRECTORY
    core logic state panels dialogs shared app src
  DESTINATION "${OMAFILES_RES_DEST}"
  FILES_MATCHING
    PATTERN "*.qml"
    PATTERN "*.js"
    PATTERN "qmldir")



# scripts/ (install-integrations.sh, dbus-*.py, open-path.sh) with execution
# permissions preserved.
install(DIRECTORY scripts DESTINATION "${OMAFILES_RES_DEST}"
        USE_SOURCE_PERMISSIONS)

# Static desktop integration uses the configured install layout. Installing the
# package only places these files; user MIME and portal defaults remain opt-in.
set(OMAFILES_GENERATED_INTEGRATIONS "${CMAKE_BINARY_DIR}/generated/integrations")
file(MAKE_DIRECTORY "${OMAFILES_GENERATED_INTEGRATIONS}")
set(OMAFILES_INTEGRATION_FILES
  io.github.percius04.omafiles.desktop
  io.github.percius04.omafiles.service
  org.freedesktop.FileManager1.service
  org.freedesktop.impl.portal.desktop.omafiles.service)
foreach(integration_file IN LISTS OMAFILES_INTEGRATION_FILES)
  configure_file(
    "${CMAKE_CURRENT_SOURCE_DIR}/integrations/${integration_file}.in"
    "${OMAFILES_GENERATED_INTEGRATIONS}/${integration_file}"
    @ONLY)
endforeach()
configure_file(
  "${CMAKE_CURRENT_SOURCE_DIR}/integrations/omafiles.portal"
  "${OMAFILES_GENERATED_INTEGRATIONS}/omafiles.portal"
  COPYONLY)

install(FILES "${OMAFILES_GENERATED_INTEGRATIONS}/io.github.percius04.omafiles.desktop"
  DESTINATION "${OMAFILES_DATA_INSTALL_DIR}/applications")
# FileManager1 is a shared well-known service name. Nautilus owns the same
# system descriptor path, so keep OmaFiles' configured copy in resources and
# place a user descriptor only during explicit --enable.
install(FILES
  "${OMAFILES_GENERATED_INTEGRATIONS}/io.github.percius04.omafiles.service"
  "${OMAFILES_GENERATED_INTEGRATIONS}/org.freedesktop.impl.portal.desktop.omafiles.service"
  DESTINATION "${OMAFILES_DATA_INSTALL_DIR}/dbus-1/services")
install(FILES "${OMAFILES_GENERATED_INTEGRATIONS}/omafiles.portal"
  DESTINATION "${OMAFILES_DATA_INSTALL_DIR}/xdg-desktop-portal/portals")
# Keep configured copies beside the resource script so a user install can
# repair a deleted descriptor. A /usr install is only checked, never shadowed.
install(DIRECTORY "${OMAFILES_GENERATED_INTEGRATIONS}/"
  DESTINATION "${OMAFILES_RES_DEST}/integrations")

# assets/ (source icon and variants).
install(DIRECTORY assets DESTINATION "${OMAFILES_RES_DEST}")

# Icons in the user's hicolor theme, so they appear in the menu / dock /
# taskbar / xdg-open / FileManager1 (Icon=omafiles in the .desktop).
#   - omafiles.svg          : main icon (light background + dark folder).
#   - omafiles-symbolic.svg : symbolic variant (currentColor, recolorable
#                             by GTK/Qt according to the theme); same geometry.
#   - omafiles-<n>.png      : rasterizations for the hicolor sizes.
set(OMAFILES_ICON_DIR "${OMAFILES_DATA_INSTALL_DIR}/icons/hicolor")
install(FILES assets/omafiles.svg assets/omafiles-symbolic.svg
  DESTINATION "${OMAFILES_ICON_DIR}/scalable/apps")
foreach(sz 32 48 64 128 256)
  install(FILES "assets/omafiles-${sz}.png"
    DESTINATION "${OMAFILES_ICON_DIR}/${sz}x${sz}/apps"
    RENAME omafiles.png)
endforeach()

# Convenience gate: build the binary, then use the same CTest definitions as CI.
add_custom_target(selfcheck
  COMMAND "${CMAKE_CTEST_COMMAND}" --test-dir "${CMAKE_BINARY_DIR}"
          --output-on-failure -R "^selfcheck\\."
  DEPENDS omafiles-standalone
  USES_TERMINAL
  COMMENT "Running source-tree and installed-tree selfchecks through CTest")

add_custom_target(asan-check
  COMMAND bash "${CMAKE_CURRENT_SOURCE_DIR}/tests/run-asan.sh"
          "${CMAKE_BINARY_DIR}/asan"
  WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}"
  USES_TERMINAL
  COMMENT "Running native safety tests under AddressSanitizer")
